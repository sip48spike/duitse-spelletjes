<script>

let score=0;
let tijd=60;

const vragen=[

{
vraag:"Wat betekent der Montag?",
antwoorden:["maandag","woensdag","vrijdag","zondag"],
juist:"maandag"
},

{
vraag:"Wat betekent die Apotheke?",
antwoorden:["kerk","apotheek","hotel","bank"],
juist:"apotheek"
},

{
vraag:"Wat betekent der Bahnhof?",
antwoorden:["restaurant","station","metro","trein"],
juist:"station"
},

{
vraag:"Wat betekent das Hotel?",
antwoorden:["hotel","kerk","bank","trein"],
juist:"hotel"
},

{
vraag:"Wat betekent das Rathaus?",
antwoorden:["bibliotheek","marktplaats","stadhuis","station"],
juist:"stadhuis"
},

{
vraag:"Wat betekent die Kirche?",
antwoorden:["kerk","apotheek","bank","metro"],
juist:"kerk"
}

];

function nieuweVraag(){

let q=vragen[Math.floor(Math.random()*vragen.length)];

document.getElementById("question").innerText=q.vraag;

let knoppen="";

q.antwoorden.forEach(a=>{
knoppen+=`
<button onclick="controleer('${a}','${q.juist}')">
${a}
</button>
`;
});

document.getElementById("answers").innerHTML=knoppen;
}

function controleer(antwoord,juist){

if(antwoord===juist){

score++;

document.getElementById("score").innerText=
"Score: "+score;

document.getElementById("result").innerHTML=
"✅ Juist!";

}
else{

document.getElementById("result").innerHTML=
"❌ Fout!";

}

nieuweVraag();
}

nieuweVraag();

let interval=setInterval(function(){

tijd--;

document.getElementById("timer").innerText=
"Tijd: "+tijd;

if(tijd<=0){

clearInterval(interval);

document.body.innerHTML=
`
<h1>🏆 Einde</h1>
<h2>Jouw score: ${score}</h2>
`;

}

},1000);

</script>
